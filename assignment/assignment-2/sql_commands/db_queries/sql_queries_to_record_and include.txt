join airlinedb_ek_updated.aircraft ON airlinedb_ek_updated.maintenance.aircraft_ID=airlinedb_ek_updated.aircraft.aircraft_ID
group by airlinedb_ek_updated.aircraft.aircraft_model_ID
order by num_of_maintenance_events  DESC

    FROM
        (((flights
        JOIN bookings ON ((flights.flight_ID = bookings.flight_ID)))
